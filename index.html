create table vehicles( id text primary key, name text not null, plate text not null, active boolean not null default true, depart_time text not null default '7:00 p. m.' check (depart_time in ('6:00 p. m.','7:00 p. m.','8:00 p. m.','9:00 p. m.')) ); -- crea el perfil al registrarse (el celular viene del correo interno celular@clientes.taxiancash.pe) create function handle_new_user() returns trigger language plpgsql security definer set search_path = public as $$ declare p text := split_part(new.email, '@', 1); begin if p !~ '^\d{9}$' then raise exception 'Celular inválido'; end if; insert into profiles(id, name, phone) values (new.id, left(coalesce(nullif(trim(new.raw_user_meta_data->>'name'), ''), 'Cliente'), 80), p); return new; end $$; create trigger on_signup after insert on auth.users for each row execute function handle_new_user(); -- el servidor calcula precios y estado: el cliente no puede manipularlos create function price_booking() returns trigger language plpgsql security definer set search_path = public as $$ declare v vehicles; h int; dep timestamptz; late boolean; begin select * into v from vehicles where id = new.vehicle_id and active; if not found then raise exception 'Unidad no disponible'; end if; if new.trip_date > current_date + 30 then raise exception 'Solo puedes reservar hasta 30 días antes'; end if; h := case v.depart_time when '6:00 p. m.' then 18 when '8:00 p. m.' then 20 when '9:00 p. m.' then 21 else 19 end; dep := (new.trip_date + make_time(h, 0, 0)) at time zone 'America/Lima'; if dep < now() then raise exception 'La hora de salida ya pasó'; end if; late := dep - now() <= interval '6 hours'; new.user_id := auth.uid(); new.seat_price := case when late then 72 else 60 end; new.total := new.seat_price + case when new.drinks then 40 else 0 end; new.deposit := round(new.total * 0.2, 2); new.balance := new.total - new.deposit; new.status := 'Pendiente'; return new; end $$; create trigger price_booking before insert on bookings for each row execute function price_booking();
